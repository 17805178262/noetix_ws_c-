# 独立 C++ 部署工程中的 noetix_policy 包。
# 默认构建 ROS2 节点，也支持 NOETIX_BUILD_ROS=OFF 以编译独立算法库。
# 在与 Python 工程完全不同的 ROS2 工作区中构建，安装正常 ament 包索引。
# 使用Release优化但不启用fast-math、量化或强制本机指令集，兼容x86与Jetson aarch64。
cmake_minimum_required(VERSION 3.16)
project(noetix_policy VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(NOETIX_BUILD_ROS "Build ROS2 nodes and ament package" ON)
# 官方 ONNX Runtime 1.23.2 CMake Config。
# x86_64 / aarch64 均在 /opt/onnxruntime 安装对应 CPU SDK。
# 不依赖工程内 Find 模块或手写 Config 文件；无须 shell 环境变量。
list(PREPEND CMAKE_PREFIX_PATH "/opt/onnxruntime")
find_package(Eigen3 REQUIRED)
find_package(onnxruntime CONFIG REQUIRED)
if(NOT TARGET onnxruntime::onnxruntime)
  message(FATAL_ERROR
    "Official onnxruntime Config loaded but target onnxruntime::onnxruntime is missing")
endif()
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(MINIZIP REQUIRED IMPORTED_TARGET minizip)
add_library(noetix_policy_core STATIC
  src/reference.cpp src/motion.cpp src/core.cpp src/inference.cpp src/registry.cpp
  src/policies/walk.cpp src/policies/gmt.cpp src/policies/motion.cpp)
target_include_directories(noetix_policy_core PUBLIC include)
target_link_libraries(noetix_policy_core PUBLIC Eigen3::Eigen onnxruntime::onnxruntime OpenSSL::Crypto Threads::Threads PkgConfig::MINIZIP)
target_compile_options(noetix_policy_core PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
if(NOETIX_BUILD_ROS)
  find_package(ament_cmake REQUIRED)
  find_package(ament_index_cpp REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(noetix_msgs REQUIRED)
  find_package(sensor_msgs REQUIRED)
  find_package(nav_msgs REQUIRED)
  find_package(std_msgs REQUIRED)
  find_package(yaml-cpp REQUIRED)
  add_library(noetix_policy_ros STATIC src/ros_policy.cpp src/policy_switch.cpp src/resources.cpp)
  target_include_directories(noetix_policy_ros PUBLIC include)
  target_link_libraries(noetix_policy_ros noetix_policy_core yaml-cpp)
  ament_target_dependencies(noetix_policy_ros rclcpp ament_index_cpp noetix_msgs sensor_msgs nav_msgs std_msgs)
  target_compile_options(noetix_policy_ros PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
  # 官方的导入库目标指向 SDK 中真实的共享库位置，用于部署运行时 RPATH。
  get_target_property(_ort_library onnxruntime::onnxruntime IMPORTED_LOCATION_RELEASE)
  if(NOT _ort_library OR _ort_library MATCHES "-NOTFOUND$")
    get_target_property(_ort_library onnxruntime::onnxruntime IMPORTED_LOCATION)
  endif()
  if(NOT _ort_library OR _ort_library MATCHES "-NOTFOUND$" OR NOT EXISTS "${_ort_library}")
    message(FATAL_ERROR "onnxruntime imported target has no valid runtime library path")
  endif()
  get_filename_component(ORT_LIBRARY_DIR "${_ort_library}" DIRECTORY)
  foreach(entry policy_node walk_policy gmt_policy motion_policy)
    add_executable(${entry} src/main.cpp)
    target_link_libraries(${entry} noetix_policy_ros)
    if(NOT entry STREQUAL "policy_node")
      string(REPLACE "_policy" "" kind "${entry}")
      target_compile_definitions(${entry} PRIVATE NOETIX_STANDALONE_KIND="${kind}")
    endif()
    set_target_properties(${entry} PROPERTIES INSTALL_RPATH "${ORT_LIBRARY_DIR}")
    install(TARGETS ${entry} DESTINATION lib/${PROJECT_NAME})
  endforeach()
  install(DIRECTORY launch config models motion DESTINATION share/${PROJECT_NAME})
  # ament_package() 自动安装正式的 noetix_policy 包清单和 ament 索引。
  ament_package()
endif()
