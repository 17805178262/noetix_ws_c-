function(enable_device name folder library source)
  set(GMR_${name}_SDK_ROOT "${THIRD_PARTY}/${folder}" CACHE PATH "${name} SDK root containing include/ and lib/")
  set(GMR_WITH_${name} AUTO CACHE STRING "${name} device support: AUTO, ON or OFF")
  set_property(CACHE GMR_WITH_${name} PROPERTY STRINGS AUTO ON OFF)
  string(TOUPPER "${GMR_WITH_${name}}" mode)
  if(NOT mode MATCHES "^(AUTO|ON|OFF)$")
    message(FATAL_ERROR "GMR_WITH_${name} must be AUTO, ON or OFF")
  endif()
  if(mode STREQUAL "OFF")
    message(STATUS "${name}: disabled by configuration")
    return()
  endif()
  set(root "${GMR_${name}_SDK_ROOT}")
  gmr_select_library("${root}" "${library}" selected)
  gmr_library_matches("${selected}" matches)
  if(NOT matches)
    if(mode STREQUAL "ON")
      message(FATAL_ERROR "${name}: SDK missing or wrong architecture for ${GMR_ARCH}: ${selected}. Set GMR_${name}_SDK_ROOT to the vendor ${GMR_ARCH} SDK, or GMR_WITH_${name}=OFF/AUTO")
    endif()
    message(WARNING "${name}: unavailable on ${GMR_ARCH}; SDK missing or wrong architecture: ${selected}. Device disabled. Supply GMR_${name}_SDK_ROOT and set GMR_WITH_${name}=ON to enable it.")
    return()
  endif()
  message(STATUS "${name}: enabled (${selected})")
    add_library(sdk_${name} SHARED IMPORTED)
    set_target_properties(sdk_${name} PROPERTIES
      IMPORTED_LOCATION "${selected}"
      INTERFACE_INCLUDE_DIRECTORIES "${root}/include")
    target_sources(retarget_node PRIVATE "readers/${source}")
    target_link_libraries(retarget_node sdk_${name})
    target_compile_definitions(retarget_node PRIVATE GMR_WITH_${name}=1)
    # Only install the selected directory, not another architecture's subdirectory.
    get_filename_component(selected_dir "${selected}" DIRECTORY)
    file(GLOB companions "${selected_dir}/*.so*")
    foreach(companion IN LISTS companions)
      gmr_require_library("${companion}" "${name} companion library")
      install(FILES "${companion}" DESTINATION lib/${PROJECT_NAME}/sdk)
    endforeach()
endfunction()
